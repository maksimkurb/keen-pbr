{{- /* Screenshots of the Web UI from static/images/changelog//. Files are named -{before,after}-{desktop,mobile}.png. With a "before" screenshot the shortcode shows before/after pairs. With only="after" it shows the desktop and mobile versions in one row, keeping their aspect ratios. @param name file name prefix @param alt alt text @param mobile "true" to add phone-size screenshots @param only "after" when there is no "before" screenshot @param dir version folder, default 3.4.0 @param names comma-separated names of "after" screenshots to show side by side at their natural size @param alts pipe-separated alt texts for names @example {{< shot name="dashboard" alt="Dashboard" mobile="true" >}} */ -}} {{- $name := .Get "name" -}} {{- $alt := .Get "alt" | default $name -}} {{- $dir := .Get "dir" | default "3.4.0" -}} {{- $onlyAfter := eq (.Get "only") "after" -}} {{- $mobile := eq (.Get "mobile") "true" -}} {{- $base := printf "images/changelog/%s/%s" $dir $name -}}
