      Полный пример конфигурации

      • Open in ChatGPT

      На этой странице приведён один полный пример конфигурации с комментариями, включая отмеченные в примере платформенные настройки. Перед применением удалите настройки, которые не поддерживаются целевой платформой.

      В config.json можно писать комментарии //. Перед применением замените примеры интерфейсов, адресов и путей к файлам на значения своей сети.

      Имена списков, теги outbound и теги DNS-серверов должны соответствовать шаблону ^[a-z][a-z0-9_]*$ и не превышать 24 символа.
      config.json
      {
  // Версия keen-pbr, для которой записан этот конфиг (X.Y.Z).
  // Ставится сама при сохранении в веб-интерфейсе; при ручной правке оставьте как есть.
  "version": "3.4.0",

  // Необязательное имя устройства в заголовке браузера и под логотипом keen-pbr.
  // Максимальная длина: 128 символов. По умолчанию: пустая строка
  // (используется стандартный брендинг keen-pbr).
  "device_name": "Домашний роутер",

  // Общие настройки сервиса keen-pbr.
  // Все поля в этом разделе необязательны.
  "daemon": {
    // Путь к PID-файлу (файлу с номером процесса сервиса).
    // По умолчанию: файл не создаётся, если поле опущено.
    "pid_file": "/var/run/keen-pbr.pid",

    // Каталог для кэша удалённых списков и метаданных кэша.
    // По умолчанию: "/var/cache/keen-pbr".
    "cache_dir": "/var/cache/keen-pbr",

    // Какой firewall использовать.
    // Поддерживаемые значения: "auto", "iptables", "nftables".
    // По умолчанию: "auto".
    "firewall_backend": "auto",

    // Не обрабатывать пакеты, которые уже помечены другой программой (fwmark).
    // По умолчанию: true (также если задано null).
    "skip_marked_packets": true,

    // Очищать динамические наборы (IP-адреса доменов из DNS-ответов и соединений)
    // при каждом применении конфигурации или перезапуске сервиса. По умолчанию: false (также если null).
    "clear_dynamic_sets_on_apply": false,

    // Необязательный начальный размер хеш-таблицы ipset, создаваемых бэкендом iptables.
    // Не действует с nftables. Минимум: 1; максимум: 2147483648. Оставьте null
    // или опустите поле для значения по умолчанию (1024).
    // При изменении наборы пересоздаются, а динамические наборы очищаются.
    "ipset_hashsize": null,

    // Необязательное максимальное число элементов ipset, создаваемых бэкендом iptables.
    // Не действует с nftables. Минимум: 1; максимум: 4294967295. Оставьте null
    // или опустите поле для значения по умолчанию (65536).
    // При изменении наборы пересоздаются, а динамические наборы очищаются.
    "ipset_maxelem": null,

    // Не перезаполнять наборы списков при SIGUSR1, переключении urltest/icmptest
    // и изменении интерфейсов: обновляются только правила. По умолчанию: true.
    "reuse_static_sets_on_runtime_refresh": true,

    // Маршрутизировать IPv6-трафик. По умолчанию: true (также если задано null).
    // Если в системе нет IPv6, keen-pbr запишет ошибку в журнал и продолжит
    // работу только с IPv4.
    "ipv6_enabled": true,

    // Блокировать трафик, если interface outbound недоступен, вместо того чтобы
    // пускать его через обычное соединение. По умолчанию: false.
    "strict_enforcement": false,

    // Что делать с трафиком, когда сработал strict_enforcement.
    // Допустимые значения: "unreachable" (вернуть сетевую ошибку) или
    // "blackhole" (молча отбросить пакеты). По умолчанию: "unreachable".
    "strict_enforcement_action": "unreachable",

    // Максимальный допустимый размер загружаемого удалённого контента, например URL-списков.
    // По умолчанию: 8388608 байт (8 MiB).
    "max_file_size_bytes": 8388608,

    // Сколько байт вывода iptables/nft читать при проверке правил firewall
    // (`keen-pbr status`, страница состояния). 0 — без ограничения.
    // По умолчанию: 262144.
    "firewall_verify_max_bytes": 262144,

    // Сколько секунд ждать вспомогательные скрипты keen-pbr (например, скрипт
    // для dnsmasq), прежде чем остановить их. Минимум: 1. По умолчанию: 30.
    "exec_timeout_seconds": 30,

    // Сколько секунд дать скрипту на завершение после остановки, прежде чем
    // убить его принудительно. Минимум: 0. По умолчанию: 2.
    "exec_kill_grace_seconds": 2
  },

  // Настройки встроенного HTTP API и веб-интерфейса.
  // ВНИМАНИЕ: этот раздел недоступен в пакете keen-pbr-headless
  "api": {
    // Включить или выключить HTTP API и веб-интерфейс.
    // По умолчанию: false.
    "enabled": true,

    // Адрес и порт, которые слушает API-сервер.
    // По умолчанию: "0.0.0.0:12121".
    "listen": "0.0.0.0:12121",

    // Максимальный размер тела HTTP-запроса в байтах. Минимум: 1024.
    // По умолчанию: 1048576.
    "max_request_body_bytes": 1048576,

    // Сколько секунд ждать HTTP-запрос от клиента.
    // Минимум: 1. По умолчанию: 15.
    "read_timeout_seconds": 15,

    // Сколько секунд отправлять HTTP-ответ клиенту.
    // Минимум: 1. По умолчанию: 15.
    "write_timeout_seconds": 15,

    // Сколько секунд держать соединение открытым между запросами.
    // Минимум: 1. По умолчанию: 20.
    "keep_alive_timeout_seconds": 20,

    // Вход по паролю для API и веб-интерфейса. По умолчанию отключён.
    // При enabled: true нужен password_hash: его создаёт команда
    // `keen-pbr hash-password --update` или веб-интерфейс (Безопасность).
    "authentication": {
      "enabled": false
    },

    // Адреса сайтов (origin), которым разрешено обращаться к API из браузера.
    // Только точные адреса http:// или https:// без пути и масок.
    // Если CORS не нужен, этот объект можно опустить.
    "cors": {
      "allowed_origins": ["https://router.example"]
    }
  },

  // Все поддерживаемые типы outbounds.
  // Outbound — канал, через который пойдёт трафик: интерфейс, таблица маршрутизации,
  // блокировка и т. д. Теги outbounds используются в правилах маршрутизации,
  // в detour DNS-серверов и в detour списков.
  // Поле "type" обязательно. Поддерживаются: interface, table, blackhole,
  // ignore, urltest и icmptest.
  "outbounds": [
    {
      // "interface" отправляет трафик через конкретный сетевой интерфейс.
      "type": "interface",

      // Уникальный тег outbound.
      // По умолчанию: нет, поле обязательно.
      "tag": "vpn",

      // Имя исходящего интерфейса.
      // По умолчанию: нет, обязательно для type="interface".
      "interface": "wg0",

      // Необязательный IPv4-шлюз для interface outbound. Укажите "auto",
      // чтобы выбрать единственный подходящий маршрут по умолчанию с
      // наименьшей метрикой в основной таблице; без поля маршрут будет без шлюза.
      // По умолчанию: null
      "gateway": "10.8.0.1",

      // Необязательный IPv6-шлюз для interface outbound. Укажите "auto",
      // чтобы выбрать единственный подходящий маршрут по умолчанию с
      // наименьшей метрикой в основной таблице; без поля маршрут будет без шлюза.
      // По умолчанию: null
      "gateway6": "2001:db8::1",

      // Своё значение strict_enforcement для этого outbound.
      // Если указано, имеет приоритет над daemon.strict_enforcement.
      // По умолчанию: наследуется daemon.strict_enforcement.
      "strict_enforcement": true,

      // Своё значение daemon.strict_enforcement_action для этого outbound.
      // Допустимые значения: "unreachable" или "blackhole".
      // По умолчанию: наследуется daemon.strict_enforcement_action.
      "strict_enforcement_action": "unreachable"
    },
    
    {
      // Ещё один interface outbound, например для обычного выхода в интернет (WAN).
      "type": "interface",
      "tag": "wan",
      "interface": "eth0",
      // Укажите шлюз явно или "auto", чтобы он определялся автоматически по
      // маршруту по умолчанию в основной таблице.
      "gateway": "auto"
    },
    
    {
      // "table" использует уже существующую таблицу маршрутизации ядра.
      "type": "table",

      // Уникальный тег outbound.
      "tag": "wan_as_table",

      // ID существующей таблицы маршрутизации Linux.
      // По умолчанию: нет, обязательно для type="table".
      "table": 254
    },

    {
      // "blackhole" отбрасывает совпавший трафик.
      "type": "blackhole",

      // Уникальный тег outbound.
      "tag": "block"
    },

    {
      // "ignore" оставляет совпавший трафик на обычной системной маршрутизации,
      // keen-pbr его не трогает.
      "type": "ignore",

      // Уникальный тег outbound.
      "tag": "direct"
    },

    {
      // "urltest" проверяет несколько кандидатов и автоматически выбирает один outbound.
      "type": "urltest",

      // Уникальный тег outbound.
      "tag": "auto_select",

      // URL для проверки задержки / доступности.
      // По умолчанию: нет, обязательно для type="urltest".
      "url": "https://www.gstatic.com/generate_204",

      // Интервал проверок в миллисекундах.
      // По умолчанию: 180000 для urltest; 60000 для icmptest.
      "interval_ms": 180000,

      // Таймаут каждой отдельной попытки проверки в миллисекундах.
      // По умолчанию: 5000 для urltest; 1000 для icmptest.
      // Не указывайте поле или задайте null, чтобы использовать значение по умолчанию для типа outbound.
      "probe_timeout_ms": 5000,

      // Не переключаться, если новый кандидат лишь немного лучше текущего.
      // Минимум: 0 мс. Для urltest по умолчанию: 100.
      "tolerance_ms": 100,

      // Стратегия: "priority" (по умолчанию) выбирает один канал.
      // "balance" (не на Keenetic) распределяет новые соединения
      // между рабочими каналами первой рабочей группы пропорционально их weight.
      "strategy": "priority",

      // Оставлено для совместимости со старыми конфигами и ни на что не влияет:
      // urltest и так блокирует трафик, если ни один канал не работает.
      "strict_enforcement": true,

      // Что делать с открытыми conntrack-соединениями, когда выбран другой канал.
      // Допустимые значения: "preserve" (по умолчанию) или "delete".
      // Если выбранный канал перестал работать, записи удаляются всегда.
      "conntrack_on_switch": "preserve",

      // Упорядоченные группы outbounds.
      // По умолчанию: нет, обязательно для type="urltest".
      // Группы проверяются в порядке массива (первая — первой).
      "outbound_groups": [
        {
          // Члены группы. Поддерживаемые типы: interface, table.
          "members": [
            // "weight" (1-100, по умолчанию 1) — доля новых соединений при
            // стратегии "balance"; при "priority" игнорируется.
            { "outbound": "vpn", "weight": 7 },
            { "outbound": "wan_as_table", "weight": 3 }
          ]
        },
        {
          // Если первая группа не работает, keen-pbr переключится на эту.
          // Если не работает ни один канал, трафик блокируется.
          "members": [{ "outbound": "wan" }]
        }
      ],

      // Поведение повторных проверок.
      "retry": {
        // Сколько раз пробовать проверку, включая первую попытку.
        // По умолчанию: 1.
        "attempts": 3,

        // Задержка между повторами в миллисекундах.
        // По умолчанию: 1000.
        "interval_ms": 1000
      },

      // Настройки circuit breaker.
      "circuit_breaker": {
        // Circuit breaker временно отключает нестабильный outbound.
        // Сколько ошибок подряд нужно, чтобы он открылся (outbound отключён).
        // Минимум: 1; максимум: 20. По умолчанию: 5.
        "failure_threshold": 5,

        // Сколько успешных проверок нужно, чтобы вернуться в нормальное состояние.
        // Минимум: 1; максимум: 20. По умолчанию: 2.
        "success_threshold": 2,

        // Через сколько миллисекунд после открытия (open) outbound получает
        // пробную проверку (half-open). Минимум: 1; максимум: 86400000.
        // По умолчанию: 30000 для urltest; для icmptest — большее из 60000 и interval_ms.
        "timeout_ms": 30000,

        // Сколько пробных проверок разрешено в состоянии half-open.
        // Минимум: 1; максимум: 10. По умолчанию: 1.
        "half_open_max_requests": 1
      }
    },

    {
      // "icmptest" отправляет ICMP Echo через кандидатов и автоматически
      // выбирает здоровый outbound.
      "type": "icmptest",
      "tag": "auto_ping",

      // Интервал проверок в миллисекундах. По умолчанию: 60000.
      "interval_ms": 60000,

      // Число ICMP Echo-запросов для каждого кандидата за один запуск.
      // Минимум: 1; максимум: 10. По умолчанию: 3.
      "count": 3,

      // Максимум неудачных пакетов в успешном запуске. Минимум: 0.
      // Значение должно быть меньше count. По умолчанию: 0.
      "max_failed": 0,

      // Пауза между последовательными ICMP-попытками. Минимум: 100 мс;
      // максимум: 1000 мс. По умолчанию: 200.
      "packet_interval_ms": 200,

      // Таймаут каждой ICMP-попытки. Рабочий диапазон: 100–5000 мс.
      // По умолчанию: 1000.
      "probe_timeout_ms": 1000,

      // Ответы медленнее этого значения считаются неудачными. Минимум: 1 мс;
      // значение не должно превышать probe_timeout_ms. По умолчанию: 500.
      "max_rtt_ms": 500,

      // Не переключаться, если новый кандидат лишь немного лучше.
      // Минимум: 0 мс. По умолчанию: 10.
      "tolerance_ms": 10,

      // Сохранять или удалять открытые conntrack-соединения, когда выбран
      // другой канал. Допустимые значения: "preserve" (по умолчанию)
      // или "delete".
      "conntrack_on_switch": "preserve",

      // Упорядоченные группы (первая проверяется первой) явных членов
      // outbound/target. Обязательно для type="icmptest". Каждый член требует
      // "outbound" и "target"; "weight" (1-100, по умолчанию 1) необязателен.
      "outbound_groups": [
        {
          "members": [
            {
              // Тег outbound типа interface или table.
              "outbound": "vpn",
              // IPv4- или IPv6-адрес для ping через этот outbound.
              "target": "1.1.1.1"
            },
            {
              "outbound": "wan",
              "target": "9.9.9.9",
              "weight": 3
            }
          ]
        }
      ]
    }
  ],

  // Список может использовать один или несколько источников: url, domains, ip_cidrs, file.
  "lists": {

    "inline_domains": {
      // Домены, заданные прямо в конфиге.
      // Совпадают и с самим доменом, и с его поддоменами.
      // Шаблоны вида "*.example.com" необязательны: "example.com" работает так же.
      "domains": ["example.com", "othersite.net"],

      // Минимальное время жизни IP-адресов, полученных перехватом DNS и L7.
      // Если поле опущено, равно 0 или меньше 1000 мс, используется
      // intercept.min_ttl_ms. Значение от 1000 мс округляется вниз до секунд.
      // TTL DNS-записи может увеличить срок до intercept.max_ttl_ms. Это не
      // интервал обновления удалённых списков.
      "ttl_ms": 86400000
    },

    "inline_ips": {
      // IPv4/IPv6-адреса или CIDR, заданные прямо в конфиге.
      "ip_cidrs": [
        "93.184.216.34",
        "10.0.0.0/8",
        "2001:db8::1",
        "2001:db8::/32"
      ]
    },

    "remote_list": {
      // URL удалённого списка.
      "url": "https://raw.githubusercontent.com/v2fly/domain-list-community/refs/heads/master/data/apple",

      // Необязательный outbound, через который будет загружаться этот список.
      // Подходят outbounds типов interface, table, urltest, icmptest.
      // По умолчанию: null (используется обычная системная маршрутизация)
      "detour": "auto_select",

      // Минимальное время жизни IP-адресов, полученных перехватом DNS и L7.
      // Если поле опущено, равно 0 или меньше 1000 мс, используется
      // intercept.min_ttl_ms. Значение от 1000 мс округляется вниз до секунд.
      // TTL DNS-записи может увеличить срок до intercept.max_ttl_ms. Это не
      // интервал обновления удалённых списков.
      "ttl_ms": 86400000
    },
  
    "local_file_list": {
      // Путь к локальному файлу списка.
      "file": "/etc/keen-pbr/local.lst",
      
      // Минимальное время жизни IP-адресов, полученных перехватом DNS и L7.
      // Если поле опущено, равно 0 или меньше 1000 мс, используется

      Примечания

      • dns.servers[].detour поддерживает outbounds типов interface, table, urltest и icmptest, но не blackhole и не ignore.
      • lists[].detour полезен, когда удалённый список нужно загружать через VPN или другой нестандартный маршрут.
      • route.rules[] должен содержать хотя бы одно условие совпадения: list, dscp, src_port, dest_port, src_addr, dest_addr или default_gateway.
      • dns.rules[].allow_domain_rebinding в основном нужен для внутренних доменов, которые специально резолвятся в приватные IP-адреса.
